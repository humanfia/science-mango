theorem M7.RecipeSignature.source_orbit_quotient : ∀ (N : ℕ) [NeZero N], ∀ c : M7.Action.Recipe N, ∀ (E : M6.Cyclic.BinaryPolynomial → Prop) (L R : Finset (ZMod N) → Prop), M7.RecipeSignature.sourceCount c E L R = M7.ActualOrbit.distinctCount c (fun y => E (M7.RecipeSignature.signature y) ∧ L y.1 ∧ R y.2) ∧ M7.ActualFactorized.stabilizerNumerator c ∣ M7.RecipeSignature.sourceNumerator c E L R := by
  intro N inst c E L R
  have h := M7.ActualFactorized.exact_orbit_quotient N c
    (M7.RecipeSignature.region E) L R
    (M7.RecipeSignature.translation_invariant N E)
  simpa only [M7.RecipeSignature.sourceCount,
    M7.RecipeSignature.sourceNumerator, M7.RecipeSignature.region,
    M7.RecipeSignature.outer_source_signature] using h
