\subsection*{Derivation bridges}

\begin{lemma}[Dry air carries the whole pressure at the reference state]
\label{lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:dryAirPartialPressure_at_T0}
\lean{IPhO2026_4_B_4.VaporPressureB4Data.dryAirPartialPressure_at_T₀}
\uses{def:IPhO2026Problems_problem_IPhO_2026_4_B_4:VaporPressureB4Data}
At the reference state $(T_0, H_0)$, with $0 < T_0$, the dry-air partial
pressure equals the full total pressure:
$p_{\mathrm{air}}(T_0, H_0) = P_{atm}$.
\end{lemma}
\begin{proof}
Dalton's law at $(T_0, H_0)$ gives
$p_{\mathrm{air}}(T_0, H_0) + P_v(T_0, H_0) = P_{atm}$, and the B.4
hypothesis $P_v(T_0, H_0) = 0$ collapses the sum.
\end{proof}

\begin{lemma}[Combined gas-law form at an admitted state]
\label{lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:total_pressure_mul_volume}
\lean{IPhO2026_4_B_4.VaporPressureB4Data.total_pressure_mul_volume}
\uses{def:IPhO2026Problems_problem_IPhO_2026_4_B_4:VaporPressureB4Data, def:IPhO2026Problems_problem_IPhO_2026_4_B_4:gasVolume}
At every admitted state $(T', H')$ with $0 < T'$ and $0 < H'$, the total
pressure times the headspace volume equals the total trapped molar
content times $R T'$:
$P_{atm}\,(A H') = \bigl(n_{\mathrm{air}} + n_{\mathrm{vap}}(T',H')\bigr)\, R\, T'$.
\end{lemma}
\begin{proof}
Rewrite $P_{atm}$ via Dalton's law, distribute the volume over the two
component pressures, replace each component term by its ideal-gas-law
right-hand side, and regroup.
\end{proof}

\begin{lemma}[Zero reference value forces identically zero vapor pressure]
\label{lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:eq_zero_of_clausiusClapeyron_zero}
\lean{IPhO2026_4_B_4.VaporPressureB4Data.eq_zero_of_clausiusClapeyron_zero}
\uses{def:IPhO2026Problems_problem_IPhO_2026_4_B_4:ClausiusClapeyron, def:IPhO2026Problems_problem_IPhO_2026_4_B_4:VaporPressureB4Data}
Consistency checkpoint between Eq.\ (3) and the B.4 zero hypothesis:
under the Clausius--Clapeyron context at reference value $P_{v0}$ for the
curve $T \mapsto P_v(T, H_0)$, if $P_{v0} = 0$ then the vapor pressure
vanishes at every temperature $T > 0$.  Context lemma for B.5/B.6; not
used by the B.4 target.
\end{lemma}
\begin{proof}
Instantiate the Clausius--Clapeyron relation at $T$ and substitute
$P_{v0} = 0$, making the right-hand side zero.
\end{proof}

\subsection*{Value and target theorems (conclusion side)}

\begin{theorem}[B.4 vapor-pressure readout formula]
\label{thm:IPhO2026Problems_problem_IPhO_2026_4_B_4:vaporPressure_eq}
\lean{IPhO2026_4_B_4.VaporPressureB4Data.vaporPressure_eq}
\leanok
\uses{def:IPhO2026Problems_problem_IPhO_2026_4_B_4:VaporPressureB4Data, def:IPhO2026Problems_problem_IPhO_2026_4_B_4:MeasuredState, lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:dryAirPartialPressure_at_T0, lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:total_pressure_mul_volume}
At every admissible measured state $(T, H)$ of the trapped dry air plus
water vapor mixture, the vapor pressure is the recorded B.4 answer
\[ P_v \;=\; P_{atm}\Bigl(1 - \frac{H_0\,T}{H\,T_0}\Bigr). \]
The Clausius--Clapeyron relation is \emph{not} assumed: B.4 uses only
Dalton's law at constant total pressure, the ideal-gas law for the fixed
dry-air content, and the vanishing of the vapor pressure at $T_0$.  The
official formula lives only here and in
\cref{thm:IPhO2026Problems_problem_IPhO_2026_4_B_4:target}, on the
conclusion side.
\end{theorem}
\begin{proof}
Apply the combined gas law at $(T, H)$ and at the reference state
$(T_0, H_0)$, where the dry air alone carries $P_{atm}$
(\cref{lem:IPhO2026Problems_problem_IPhO_2026_4_B_4:dryAirPartialPressure_at_T0}),
so the dry-air ideal-gas law gives
$P_{atm} A H_0 = n_{\mathrm{air}} R T_0$.  Eliminating the fixed dry-air
content $n_{\mathrm{air}}$ between this reference balance and the state
balance yields $p_{\mathrm{air}}(T,H)\,H T_0 = P_{atm} H_0 T$, and
subtracting from $P_{atm}$ via Dalton's law gives the claimed closed
form; the contracted sorried proof body is left to the prover stage.
\end{proof}
