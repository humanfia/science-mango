\uses{def:IPhO2026Problems_problem_IPhO_2026_1_B_2:angleBetween, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:IsAsymptoticRelativeVelocity, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:initialDirection, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:eccentricitySq, thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:orbit_eq_conic, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:CoulombScatteringData}
Any asymptotic relative velocity satisfies
$\lVert u_\infty \rVert = \sqrt{2E/m_{\mathrm{red}}}$ and
$\mathrm{angleBetween}\ \hat u_0\ u_\infty
= \pi - 2\arctan\!\big(1/\sqrt{\varepsilon^2 - 1}\big)$.
Contains the sign/orientation input but no numeric deflection value.
\end{theorem}
\begin{proof}
The magnitude follows from the energy law at infinity, where the Coulomb
term vanishes and $E = \tfrac12 m_{\mathrm{red}} \lVert u_\infty \rVert^2$.
For the direction, the hyperbola of
\cref{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:orbit_eq_conic} with
$\varepsilon > 1$ has asymptote directions making the acute angle
$\arctan\sqrt{\varepsilon^2 - 1}$ with the symmetry axis; traversed outward
from periapsis, the total change of direction between the incoming line and
the outgoing asymptote is
$\pi - 2\arctan\!\big(1/\sqrt{\varepsilon^2-1}\big)$, read off on the
branch side recorded in
\cref{def:IPhO2026Problems_problem_IPhO_2026_1_B_2:IsAsymptoticRelativeVelocity}.
\end{proof}

\begin{lemma}[Branch sign: signed deflection is the negated angle]
\label{lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:signedDeflection_eq_neg_angle}
\lean{IPhO2026.Problem1.B2.signedDeflection_eq_neg_angle}
\uses{def:IPhO2026Problems_problem_IPhO_2026_1_B_2:signedDeflection, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:IsAsymptoticRelativeVelocity}
Under the strict Fig.-1b branch condition
$\mathrm{perp}\ \hat u_0\ u_\infty < 0$, the signed deflection satisfies
$\mathrm{signedDeflection}\ u = -\mathrm{angleBetween}\ \hat u_0\ u_\infty$.
\end{lemma}
\begin{proof}
Unfold the definition; the strict negativity is precisely the failing
condition of the sign test, so the $-1$ branch is selected and the formula
becomes the negation.  (The degenerate zero-deflection case is excluded by
the physics side: the branch condition together with the nonzero
deflection from
\cref{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:signed_deflection_eq_formula}
makes the remaining branch strict.)
\end{proof}

\begin{theorem}[Main target: signed deflection angle (T1-B2)]
\label{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:signed_deflection_angle_T1_B2}
\lean{IPhO2026.Problem1.B2.signed_deflection_angle_T1_B2}
\uses{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:exists_asymptoticRelativeVelocity, thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:signed_deflection_eq_formula, lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:signedDeflection_eq_neg_angle, thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:eccentricity_sq_eq, lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:asymptote_factor_certificate, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:signedDeflection, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:radiansToDegrees, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:roundsToOfficialDegrees, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:unboundMu, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:IsAngularMomentumFactor, def:IPhO2026Problems_problem_IPhO_2026_1_B_2:CoulombScatteringData}
Under the two-body Coulomb model of Fig. 1b with $\mu = 15/2$ (unbound
case), there exists an asymptotic relative velocity $u_\infty$ of $e^+$
with respect to $e^-$, and its signed deflection $\delta$ from the initial
line of motion of $e^+$ equals the exact value
$-(\pi - 2\arctan(2/\sqrt{63}))$ radians, whose degree reading rounds to
the official $-16.60$ degrees -- the negative sign being the official
``$16.60$ degrees below the initial line of motion''.  The recorded
official value first appears here, conclusion-side.
\end{theorem}
\begin{proof}
Existence of $u_\infty$ is
\cref{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:exists_asymptoticRelativeVelocity}.
The formula
\cref{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:signed_deflection_eq_formula}
with $\varepsilon^2 = 67/4$
(\cref{thm:IPhO2026Problems_problem_IPhO_2026_1_B_2:eccentricity_sq_eq})
and the asymptote factor $2/\sqrt{63}$
(\cref{lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:asymptote_factor_certificate})
gives the unsigned angle $\pi - 2\arctan(2/\sqrt{63})$.  The deflection is
nonzero, so the branch condition sharpens to
$\mathrm{perp}\ \hat u_0\ u_\infty < 0$ and
\cref{lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:signedDeflection_eq_neg_angle}
attaches the negative sign.  Converting by
\cref{def:IPhO2026Problems_problem_IPhO_2026_1_B_2:radiansToDegrees} gives
$-(\pi - 2\arctan(2/\sqrt{63})) \cdot 180/\pi \approx -16.60$, which lies
in $[-16.605, -16.595)$ and hence rounds to $-16.60$ by
\cref{def:IPhO2026Problems_problem_IPhO_2026_1_B_2:roundsToOfficialDegrees}.
\end{proof}

\begin{lemma}[Asymptote factor certificate]
\label{lem:IPhO2026Problems_problem_IPhO_2026_1_B_2:asymptote_factor_certificate}
\lean{IPhO2026.Problem1.B2.asymptote_factor_certificate}
The hyperbola $\varepsilon^2 = 67/4$ makes the scattering asymptote factor
$1/\sqrt{\varepsilon^2 - 1} = 1/\sqrt{67/4 - 1} = 2/\sqrt{63}$.
\end{lemma}
\begin{proof}
$67/4 - 1 = 63/4 = (\sqrt{63}/2)^2$ using $\sqrt{63}^{\,2} = 63$; taking
the positive square root gives $\sqrt{63/4} = \sqrt{63}/2$, and inverting
yields $2/\sqrt{63}$.  Pure algebra up to that square-root identity.
\end{proof}
