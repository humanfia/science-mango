\subsection*{Derivation bridges}

\begin{lemma}[Impact parameter from incidence angle]
\label{lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:impactParam_eq_sin}
\lean{IPhO2026_2_B_1.impactParam_eq_sin}
\uses{def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CookerParams, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:incidenceAngle}
Inside the open aperture the impact-parameter magnitude is recovered from
the incidence angle as $|x| = R \sin\theta$; the absolute value absorbs the
two mirror-image branches $\pm|x|$ of Figure~2f.
\end{lemma}
\begin{proof}
Unfold the incidence angle $\arcsin(|x|/R) = \theta$: on the open aperture
$|x|/R < 1$ lies in the range of $\sin$ on the acute branch, so applying
$\sin$ to both sides gives $|x| = R\sin\theta$.
\end{proof}

\begin{lemma}[Positivity of $\sin(2\theta)$ on the acute branch]
\label{lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:sin_two_pos}
\lean{IPhO2026_2_B_1.sin_two_pos}
For $\theta \in (0, \pi/2)$ one has $0 < \sin(2\theta)$; the positivity
flashpoint that keeps the coefficient identification nondegenerate.
\end{lemma}
\begin{proof}
On $\theta \in (0, \pi/2)$ the double angle $2\theta$ lies in $(0, \pi)$,
where the sine is strictly positive.
\end{proof}

\begin{theorem}[Tangency relation at the limiting ray]
\label{thm:IPhO2026Problems_problem_IPhO_2026_2_B_1:container_radius_at_extremal_angle}
\lean{IPhO2026_2_B_1.container_radius_at_extremal_angle}
\uses{def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CookerParams, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CookerB1, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:ExtremalRaySpec, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:distToLine, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:incidenceAngle, lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:impactParam_eq_sin}
At an extremal angle $\theta \in (0, \pi/2)$ with extremal-ray tangency
bookkeeping, specular reflection at the mirror point
$(|x|, -\sqrt{R^2 - |x|^2})$ with $|x| = R\sin\theta$, the container offset
$A = (0, -R/2)$, and the tangency distance condition force the container
radius to be
\[
a = R \sin\theta - \tfrac{R}{2}\sin(2\theta) .
\]
This is the load-bearing geometric identity of part B.1.
\end{theorem}
\begin{proof}
The specular law at the extremal column is a two-by-two linear system for
the slope and intercept of the reflected line, with nonzero determinant
$-(2x)(x^2 + y^2)$; solving gives the intercept
$b = -R^2/(2\sqrt{R^2 - x^2})$, always negative with $|b| \ge R/2$.
The tangency distance condition then evaluates (on either mirror-image
branch, via $|x|$) since the signed-distance numerator $-R/2 - b$ is
strictly positive; substituting $|x| = R\sin\theta$ from
\cref{lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:impactParam_eq_sin}
and eliminating the double angle gives the claimed identity.
\end{proof}

\begin{theorem}[Coefficients of the given ansatz in terms of $R$]
\label{thm:IPhO2026Problems_problem_IPhO_2026_2_B_1:alpha_beta_in_terms_of_R}
\lean{IPhO2026_2_B_1.alpha_beta_in_terms_of_R}
\uses{def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CookerParams, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CookerB1, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:IsThetaMax, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:ExtremalRaySpec, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CoeffSpec, def:IPhO2026Problems_problem_IPhO_2026_2_B_1:SecondExtremalConfig, lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:sin_two_pos, thm:IPhO2026Problems_problem_IPhO_2026_2_B_1:container_radius_at_extremal_angle}
\textbf{(Target, T2-B1.)}  For the Figure-2f cooker with extremal absorbed
ray at maximum incidence angle $\theta_{\max}$, a coefficient pair
$\alpha, \beta$ fulfilling the family ansatz
\cref{def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CoeffSpec}, and a
second extremal configuration of the same mirror radius at a distinct
extremal angle $\theta' \neq \theta_{\max}$, the coefficients of the given
ansatz $a = \alpha \sin\theta_{\max} + \beta \sin(2\theta_{\max})$ are
\[
\alpha = R, \qquad \beta = -\frac{R}{2} .
\]
This is the recorded official answer of part B.1; the values are
conclusion-side only, confined to this target theorem (no hypothesis or
structure field states them in advance).
\end{theorem}
\begin{proof}
Apply
\cref{thm:IPhO2026Problems_problem_IPhO_2026_2_B_1:container_radius_at_extremal_angle}
at the given extremal ray and at the second configuration
$\theta'$: both angles satisfy the tangency identity, while the ansatz of
\cref{def:IPhO2026Problems_problem_IPhO_2026_2_B_1:CoeffSpec} gives the
family identity at the same two configurations.  Subtracting at
$\theta_{\max}$ and $\theta'$ yields a two-by-two homogeneous linear
system in $(\alpha - R, \beta + R/2)$ whose determinant
$2\sin\theta_{\max}\sin\theta'\,(\cos\theta' - \cos\theta_{\max})$ is
nonzero --- both sines are positive by
\cref{lem:IPhO2026Problems_problem_IPhO_2026_2_B_1:sin_two_pos} (and
$\sin\theta > 0$ on the acute branch), and $\cos$ is strictly
antitone on $(0, \pi/2)$, so $\theta' \neq \theta_{\max}$ gives
$\cos\theta' \neq \cos\theta_{\max}$ --- forcing $\alpha = R$ and
$\beta = -R/2$.
\end{proof}
