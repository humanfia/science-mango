\begin{lemma}[Heat leaves the torus on field increase]

\lean{IPhO2026.Problem3.C4.heat_leaves_torus_on_field_increase}

increased at fixed positive temperature ($0 \le H_i < H_f$, $T > 0$),

\end{proof}

\begin{theorem}[Accumulation bridge: $t$ as the residence-density integral]

\lean{IPhO2026.Problem3.C4.elapsedTime_eq_integral}

The elapsed time of the cooling run, defined operationally as the
integral of the residence-time density over the visited temperatures
(additivity of time over the run, hypothesis \texttt{haccum}), equals

$t = \int_{T' \in [T,\,T_0]} (C_c/P)\,(T_h/T' - 1)\,dT'$.
\end{theorem}
\begin{proof}
Rewrite the integrand by the residence-density identity on the open
interval $(T, T_0)$; set-integral congruence extends the rewrite to
the closed interval since the two endpoints form a measure-zero set.
\end{proof}

\begin{theorem}[Evaluation bridge: the residence-time integral]
\label{thm:IPhO2026Problems_problem_IPhO_2026_3_C_4:cooling_time_integral_eval}
\lean{IPhO2026.Problem3.C4.cooling_time_integral_eval}
\uses{def:IPhO2026Problems_problem_IPhO_2026_3_C_4:globalQuantities, def:IPhO2026Problems_problem_IPhO_2026_3_C_4:RegimeAssumptions}
The residence-time integral evaluates in closed form:
$\int_{[T,\,T_0]} (C_c/P)\,(T_h/T' - 1)\,dT'
= (C_c/P)\,\bigl(T_h\,\ln(T_0/T) - (T_0 - T)\bigr)$.
(Conclusion-side intermediate value; not the official C.4 answer
form.)
\end{theorem}
\begin{proof}

$\int T_h/T' \, dT' = T_h \ln(T_0/T)$ off $T' \neq 0$ and
$\int 1\,dT' = T_0 - T$; pull the constant $C_c/P$ out.
\end{proof}

$T$ in elapsed time
\[
t = \frac{C_c\,T_h}{P}\Bigl(\ln\frac{T_0}{T} - \frac{T_0 - T}{T_h}\Bigr).
\]
The right-hand side is the recorded official answer of subquestion
C.4; it is purely conclusion-side --- the hypotheses only bundle the
per-cycle governing laws (the cooling-run predicate, composed of the
Carnot heat ratio, the first law, the constant-power law, and the
calorimetric law) and the operational definition of the elapsed time.
\end{theorem}
\begin{proof}
Chain the accumulation bridge with the evaluation bridge, then apply
the field algebra
$(C_c/P)\,(T_h \ln(T_0/T) - (T_0 - T))
= (C_c T_h/P)\,(\ln(T_0/T) - (T_0 - T)/T_h)$,
factoring $T_h$ out of the parenthesis.
\end{proof}
