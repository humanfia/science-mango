\begin{theorem}[Physics formalization target]
\label{thm:physics:phyx_mini_0642:target}
\lean{PhyXMiniProblems.ProblemPhyXMini0642.photoelectrons_ejected_per_second}
\uses{def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-particleratepersecond, def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-photoelectricexperiment, def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-matchesphotoelectricexperiment, def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-matchessuppliedphotoelectricgraph, def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-usesroundedelectronchargedatum, def:physics:phyx-mini-0642:phyxminiproblems-problemphyxmini0642-satisfiessaturationchargeflowlaw}
The \(10\,\mu\mathrm A\) saturation plateau corresponds to
\(6.25\times10^{13}\) emitted photoelectrons per second, answer B.
\end{theorem}
\begin{proof}
At saturation the charge-flow law is \(I=qN\).  The graph gives
\(I=10\,\mu\mathrm A=10^{-5}\,\mathrm A\), and the rounded electron-charge
datum gives \(q=1.6\times10^{-19}\,\mathrm C\).  Positivity makes \(q\)
nonzero, so
\[
 N=\frac{10^{-5}}{1.6\times10^{-19}}
  =6.25\times10^{13}\,\mathrm{s^{-1}}.
\]
\end{proof}
